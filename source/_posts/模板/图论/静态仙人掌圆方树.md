#include<bits/stdc++.h>
using namespace std;
#define endl '\n'
#define ll long long
//#define int long long
#define ld long double
#define fi first
#define se second
#define INF 1e18
#define MOD 1000000009
#define eps 1e-14
const int N=1e4+10;
const int M=3e3+10;
const int LOG=30;
ll n,m,q;
vector<pair<ll,ll>>adj[N];
vector<pair<ll,ll>>T[N];
ll tot,idx;
ll dfn[N],low[N];
ll fw[N],fap[N],dtree[N],backw[N];
ll pref[N],clen[N];
stack<ll>stk;
void tarjan(ll u,ll fa){
	dfn[u]=low[u]=++idx;
	stk.push(u);
	for(auto[v,w]:adj[u]){
		if(v==fa)continue;
		if(!dfn[v]){
			fap[v]=u;
			fw[v]=w;
			dtree[v]=dtree[u]+w;
			tarjan(v,u);
			low[u]=min(low[u],low[v]);
			if(low[v]>dfn[u]){
				T[u].push_back({v,w});
				T[v].push_back({u,w});
				stk.pop();
			}
			else if(low[v]==dfn[u]){
				ll btm=stk.top();
				ll len=dtree[btm]-dtree[u]+backw[btm];
				tot++;
				T[u].push_back({tot,0});
				T[tot].push_back({u,0});
				while(true){
					ll x=stk.top();
					stk.pop();
					ll arc=dtree[x]-dtree[u];
					pref[x]=arc;
					ll ww=min(arc,len-arc);
					T[tot].push_back({x,ww});
					T[x].push_back({tot,ww});
					if(x==v)break;
				}
				clen[tot]=len;
			}
		}
		else if(dfn[v]<dfn[u]){
			low[u]=min(low[u],dfn[v]);
			backw[u]=w;
		}
	}
}
ll dep[N],up[N][LOG],dis[N];
void build(){
	for(int i=1;i<=tot;i++){
		dep[i]=-1;
	}
	queue<ll>que;
	dep[1]=0,dis[1]=0,up[1][0]=0;
	que.push(1);
	while(!que.empty()){
		ll u=que.front();
		que.pop();
		for(auto[v,w]:T[u]){
			if(dep[v]!=-1)continue;
			dep[v]=dep[u]+1;
			dis[v]=dis[u]+w;
			up[v][0]=u;
			for(int k=1;k<LOG;k++){
				up[v][k]=up[up[v][k-1]][k-1];
			}
			que.push(v);
		}
	}
}
ll jump(ll u,ll k){
	for(int i=0;k;i++,k>>=1){
		if(k&1)u=up[u][i];
	}
	return u;
}
ll lca(ll u,ll v){
	if(dep[u]<dep[v])swap(u,v);
	u=jump(u,dep[u]-dep[v]);
	if(u==v)return u;
	for(int i=LOG-1;i>=0;i--){
		if(up[u][i]!=up[v][i]){
			u=up[u][i];
			v=up[v][i];
		}
	}
	return up[u][0];
}
int main(){
    ios::sync_with_stdio(0);
    cin.tie(0);
	cin>>n>>m>>q;
	for(int i=1;i<=m;i++){
		ll u,v,w;
		cin>>u>>v>>w;
		adj[u].push_back({v,w});
		adj[v].push_back({u,w});
	}
	tot=n;
	tarjan(1,0);
	build();
	while(q--){
		ll u,v;
		cin>>u>>v;
		ll l=lca(u,v);
		ll ans;
		if(l<=n){
			ans=dis[u]+dis[v]-2*dis[l];
		}
		else{
			ll a=jump(u,dep[u]-dep[l]-1);
			ll b=jump(v,dep[v]-dep[l]-1);
			ll d=abs(pref[a]-pref[b]);
			ans=dis[u]-dis[a]+dis[v]-dis[b]+min(d,clen[l]-d);
		}
		cout<<ans<<endl;
	}
    return 0;
}
