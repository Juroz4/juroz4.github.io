#include<bits/stdc++.h>
using namespace std;
#define endl '\n'
#define ll long long

#define ld long double
#define fi first
#define se second
#define INF 1e18
#define MOD 998244353
#define eps 1e-9
const int N=2e5+10;
vector<ll>adj[N];
ll fac[N+1],ifac[N+1];
ll qpow(ll b,ll e){
    ll res=1;
    while(e){
        if(e&1){
            res=res*b;
            res%=MOD;
        }
        b*=b;
        b=b%MOD;
        e>>=1;
    }
    return res;
}
void init(){
    fac[0]=1;
    for(int i=1;i<=N;i++){
        fac[i]=fac[i-1]*i%MOD;
    }
    ifac[N]=qpow(fac[N],MOD-2);
    for(int i=N;i>=1;i--){
        ifac[i-1]=ifac[i]*i%MOD;
    }
}
ll C(ll n,ll k){
    if(k<0||k>n)return 0;
    return fac[n]*ifac[k]%MOD*ifac[n-k]%MOD;
}
