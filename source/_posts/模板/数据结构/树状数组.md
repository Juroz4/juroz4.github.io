struct BIT{
        int n;
        vector<ll> tr;
        BIT(int _n):n(_n),tr(_n+1,0){}
        int lowbit(int x){
                return x&-x;
        }
        // 单点加：i 位置 += v
        void add(int i,ll v){
                for(;i<=n;i+=lowbit(i)){
                        tr[i]+=v;
                }
        }
        // 前缀和：[1,i] 的和
        ll sum(int i){
                ll res=0;
                for(;i>0;i-=lowbit(i)){
                        res+=tr[i];
                }
                return res;
        }
        // 区间和：[l,r] 的和
        ll query(int l,int r){
                return sum(r)-sum(l-1);
        }
  }; 两个常用扩展，需要时直接套：

  1. 区间加 + 单点查（用差分，开一个 BIT 维护差分数组）：

  // 原数组 a，对 [l,r] 整体 +v：add(l,v); add(r+1,-v);
  // 查 a[i] 的值 = sum(i)

  2. 区间加 + 区间查（两个 BIT，一个维护差分 d，一个维护 i*d[i]）：
 struct BIT2{
        int n;
        vector<ll> t1,t2;   // t1 存 d，t2 存 i*d[i]
        BIT2(int _n):n(_n),t1(_n+1,0),t2(_n+1,0){}
        int lowbit(int x){
                return x&-x;
        }
        void _add(int i,ll v){
                for(int x=i;x<=n;x+=lowbit(x)){
                        t1[x]+=v;
                        t2[x]+=v*i;
                }
        }
        ll _sum(int i){
                ll a=0,b=0;
                for(int x=i;x>0;x-=lowbit(x)){
                        a+=t1[x];
                        b+=t2[x];
                }
                return a*(i+1)-b;
        }
        void add(int l,int r,ll v){
                _add(l,v);
                _add(r+1,-v);
        }
        ll query(int l,int r){
                return _sum(r)-_sum(l-1);
        }
  };
