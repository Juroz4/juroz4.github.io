#include<bits/stdc++.h>
using namespace std;
#define endl '\n'
#define ll long long
//#define int long long
#define ld long double
#define fi first
#define se second
#define INF 1e18
#define MOD 998244353
#define eps 1e-9
const int N=2e3+10;
//vector<ll>adj[N];
struct edge{
	ll to;
	ll cap;
};
void merge(vector<vector<edge>>&adj,ll n,ll prev,ll last){
	vector<ll>newcap(n+1,0);
	vector<ll>mark(n+1,0);
	vector<ll>touched;
	for(edge&e:adj[prev]){
		ll w=e.to;
		if(w==last||w==prev){
			continue;
		}
		newcap[w]+=e.cap;
		if(!mark[w]){
			mark[w]=1;
			touched.push_back(w);
		}
	}
	for(edge&e:adj[last]){
		ll w=e.to;
		if(w==prev||w==last){
			continue;
		}
		newcap[w]+=e.cap;
		if(!mark[w]){
			mark[w]=1;
			touched.push_back(w);
		}
	}
	vector<edge>np;
	for(ll w:touched){
		np.push_back({w,newcap[w]});
	}
	adj[prev]=np;
	for(ll w:touched){
		vector<edge>nw;
		for(edge&e:adj[w]){
			if(e.to==last||e.to==prev){
				continue;
			}
			nw.push_back(e);
		}
		nw.push_back({prev,newcap[w]});
		adj[w]=nw;
	}
	adj[last].clear();
}
ll mincut(ll n,vector<vector<edge>>&adj){
	ll ans=INF;
	vector<ll>del(n+1,0);
	for(int phase=1;phase<n;phase++){
		vector<ll>dis(n+1,0);
		vector<ll>used(n+1,0);
		ll prev=-1,last=-1;
		for(int i=1;i<=n-phase+1;i++){
			ll u=-1;
			for(int j=1;j<=n;j++){
				if(!del[j]&&!used[j]&&(u==-1||dis[j]>dis[u])){
					u=j;
				}
			}
			used[u]=1;
			prev=last;
			last=u;
			for(edge&e:adj[u]){
				if(!del[e.to]&&!used[e.to]){
					dis[e.to]+=e.cap;
				}
			}
		}
		ans=min(ans,dis[last]);
		merge(adj,n,prev,last);
		del[last]=1;
	}
	return ans;
}
int main(){
	ios::sync_with_stdio(0);
	cin.tie(0);
	ll t;
	cin>>t;
	while(t--){
		ll n,m;
		cin>>n>>m;
		vector<vector<edge>>adj(n+1);
		for(int i=1;i<=m;i++){
			ll x,y,z;
			cin>>x>>y>>z;
			adj[x].push_back({y,z});
			adj[y].push_back({x,z});
		}
		if(n<=1){
			cout<<0<<endl;
			continue;
		}
		cout<<(n-1)*(mincut(n,adj))<<endl;
	}
	return 0;
}
