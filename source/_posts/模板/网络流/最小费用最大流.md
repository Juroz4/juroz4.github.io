#include<bits/stdc++.h>
using namespace std;
#define endl '\n'
#define ll long long
#define ld long double
#define fi first
#define se second
#define INF 1e18
#define MOD (ll)(1e9+7)
#define eps 1e-9
const ll N=2e4+10;

struct edge{
	int to;
	ll cap;
	ll cost;
	int rev;
};

vector<edge> g[N];
ll h[N];        // 势能
ll dist[N];     // 重标号后的最短路
int pre_v[N];   // 前驱点
int pre_e[N];   // 前驱边编号

void addedge(int u,int v,ll c,ll w){
	g[u].push_back({v,c,w,(int)g[v].size()});
	g[v].push_back({u,0,-w,(int)g[u].size()-1});
}

// Dijkstra 求重标号权值 w+h[u]-h[v] 下的最短路（势能保证非负）
bool dijkstra(int s,int t){
	priority_queue<pair<ll,int>,vector<pair<ll,int>>,greater<pair<ll,int>>> pq;
	fill(dist,dist+N,INF);
	dist[s]=0;
	pq.push({0,s});
	while(!pq.empty()){
		auto [d,u]=pq.top();
		pq.pop();
		if(d>dist[u]){
			continue;
		}
		for(int i=0;i<(int)g[u].size();i++){
			edge& e=g[u][i];
			if(e.cap>0){
				ll nd=d+e.cost+h[u]-h[e.to];
				if(nd<dist[e.to]){
					dist[e.to]=nd;
					pre_v[e.to]=u;
					pre_e[e.to]=i;
					pq.push({nd,e.to});
				}
			}
		}
	}
	return dist[t]<INF;
}

// 最小费用最大流（Dijkstra + 势能）
pair<ll,ll> min_cost_flow(int s,int t){
	ll flow=0,cost=0;
	fill(h,h+N,0);
	while(dijkstra(s,t)){
		// 更新势能：h[v] += dist[v]，使 h[t] 成为真实最短路费用
		for(int v=0;v<N;v++){
			if(dist[v]<INF){
				h[v]+=dist[v];
			}
		}
		// 回溯找瓶颈
		ll f=INF;
		for(int v=t;v!=s;v=pre_v[v]){
			f=min(f,g[pre_v[v]][pre_e[v]].cap);
		}
		// 增广
		for(int v=t;v!=s;v=pre_v[v]){
			edge& e=g[pre_v[v]][pre_e[v]];
			e.cap-=f;
			g[v][e.rev].cap+=f;
		}
		flow+=f;
		cost+=f*h[t];
	}
	return {flow,cost};
}
int main(){
    ios::sync_with_stdio(0);
	cin.tie(0);
    string t;
    cin>>t;
    ll n;
    cin>>n;
    ll tl=t.size();
    ll cnt_t[26]={0};
    for(char ch:t){
    	cnt_t[ch-'a']++;
	}
	ll source=0;
	ll sink=26+n+1;
	for(int c=0;c<26;c++){
		if(cnt_t[c]>0){
			addedge(source,1+c,cnt_t[c],0);
		}
	}
	for(int i=1;i<=n;i++){
		string s;
		ll a;
		cin>>s>>a;
		ll cnt_a[26]={0};
		for(char ch:s){
			cnt_a[ch-'a']++;
		}
		for(int c=0;c<26;c++){
			if(cnt_a[c]>0){
				addedge(1+c,26+i,cnt_a[c],i);
			}
		}
		addedge(26+i,sink,a,0);
	}
	auto [flow,cost]=min_cost_flow(source,sink);
	if(flow==tl){
		cout<<cost<<endl;
	}
	else cout<<-1<<endl;
    return 0;
}
