#include<bits/stdc++.h>
using namespace std;
#define endl '\n'
#define ll long long

#define ld long double
#define rep(i,a,b) for(int i=a;i<=b;i++)
#define fi first
#define se second
#define INF 1e18
#define MOD 998244353
#define eps 1e-16
const int N=2e5+10;
const int M=3e3+10;
const int LOG=30;
const long double pi=acosl(-1.0L);
struct edge{
	ll to;
	ll cap;
	ll cost;
	ll rev;
};
vector<edge>g[N];
ll dist[N];
ll pre_v[N];
ll pre_e[N];
ll h[N];
void addedge(ll u,ll v,ll c,ll w){
	g[u].push_back({v,c,w,g[v].size()});
	g[v].push_back({u,0,-w,g[u].size()-1});
}
bool dijkstra(ll s,ll t){
	priority_queue<pair<ll,ll>,vector<pair<ll,ll>>,greater<pair<ll,ll>>>pq;
	fill(dist,dist+N,INF);
	dist[s]=0;
	pq.push({0,s});
	while(!pq.empty()){
		auto[d,u]=pq.top();
		pq.pop();
		if(d>dist[u]){
			continue;
		}
		for(int i=0;i<g[u].size();i++){
			edge&e=g[u][i];
			if(e.cap>0){
				ll nd=d+e.cost+h[u]-h[e.to];
				if(nd<dist[e.to]){
					dist[e.to]=nd;
					pre_v[e.to]=u;
					pre_e[e.to]=i;
					pq.push({nd,e.to});
				}
			}
		}
	}
	return dist[t]<INF;
}
void spfa(ll s){
	fill(h,h+N,INF);
	queue<ll>q;
	vector<int>inq(N,0);
	h[s]=0; q.push(s); inq[s]=1;
	while(!q.empty()){
		ll u=q.front(); q.pop(); inq[u]=0;
			for(auto&e:g[u]){
			if(e.cap>0 && h[u]+e.cost<h[e.to]){
				h[e.to]=h[u]+e.cost;
				if(!inq[e.to]){ inq[e.to]=1; q.push(e.to); }
			}
		}
	}
	for(int i=0;i<N;i++) if(h[i]>=INF/2) h[i]=0;
}
pair<ll,ll>mcmf(ll s,ll t){
	ll flow=0,cost=0;
	spfa(s);
//	fill(h,h+N,INF);
	while(dijkstra(s,t)){
		for(int v=0;v<N;v++){
			if(dist[v]<INF){
				h[v]+=dist[v];
			}
		}
		ll f=INF;
		for(int v=t;v!=s;v=pre_v[v]){
			f=min(f,g[pre_v[v]][pre_e[v]].cap);
		}
		for(int v=t;v!=s;v=pre_v[v]){
			edge&e=g[pre_v[v]][pre_e[v]];
			e.cap-=f;
			g[v][e.rev].cap+=f;
		}
		flow+=f;
		cost+=f*h[t];
	}
	return {cost,flow};
}
int main(){
	ios::sync_with_stdio(0);

	ll n;
	cin>>n;
	vector<ll>a(n+1);
	for(int i=1;i<=n;i++){

	}
	ll ts=4*n+2;ll S=0;
	addedge(ts,S,4,0);
	ll T=4*n+1;
	// in i
	// out i+n
	// mod i+2n
	// dif i+3n
	for(int i=1;i<=n;i++){
		addedge(i,i+n,1,-1);
		addedge(i+2*n,i,INF,0);
		addedge(i+3*n,i,INF,0);
		addedge(S,i,INF,0);
		addedge(i+n,T,INF,0);
		for(int j=i+1;j<=n;j++)if(a[i]-a[j]==1){
			addedge(i+n,j+3*n,INF,0);
			break;
		}
		for(int j=i+1;j<=n;j++)if(a[j]-a[i]==1){
			addedge(i+n,j+3*n,INF,0);
			break;
		}
		for(int j=i+1;j<=n;j++)if(abs(a[i]-a[j])%7==0){
			addedge(i+n,j+2*n,INF,0);
//			addedge(j+n,i+2*n,INF,0);
			addedge(i+2*n,j+2*n,INF,0);
			break;
		}
		for(int j=i+1;j<=n;j++)if(a[i]==a[j]){
			addedge(i+3*n,j+3*n,INF,0);
			break;
		}
	}
	auto[cost,flow]=mcmf(ts,T);
